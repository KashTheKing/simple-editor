1//! Timeline widget (custom painted). Header column (track name, M/S buttons, height drag handle at the
2//! bottom edge), ruler (ticks, in/out markers, click/drag scrubs), track lanes (video tracks top→bottom =
3//! Vn..V1, then A1..An), clips (name, waveform for audio via WaveformCache, keyframe diamonds, disabled
4//! = dimmed, selected = accent outline), playhead (drag by ruler or line).
5//!
6//! Interaction: click select (Ctrl toggles), drag clip body = move (linked clips follow; Project::move_clips
7//! all-or-nothing; up/down across tracks of the same kind), drag clip edges = trim (Clip::trim_start /
8//! trim_end with Project::max_clip_duration), snapping (playhead, clip edges, in/out, 0; ~8 px) when
9//! `snap`, right-click context menu (Split, Delete, Ripple Delete, Link/Unlink, Enable/Disable, Add
10//! Video/Audio Track, Remove Track (on headers), Mute/Solo). Ctrl+Scroll = zoom around the cursor,
11//! Shift+Scroll = horizontal pan, Alt+Scroll = height of the track under the cursor, plain scroll =
12//! vertical pan. Accepts egui dnd payloads `DragPayload` (Asset → Project::insert_asset_clips at the drop
13//! time/track; Path → returned in `dropped_files`; Effect / Transition → the clip under the pointer is
14//! outlined while dragging and reported in `dropped_other`, so a card lands straight on that clip).
15//! Every mutation calls `c.undo(&project)` with the project as
16//! it was before the gesture (once per gesture, only if it changed something) and sets `edited`. While `playing`,
17//! auto-scroll keeps the playhead visible unless the user panned away (resumes once the playhead re-enters the
18//! view or playback pauses).
19//!
20//! Round 2: horizontal/vertical scrollbars (thumb drag + click-to-page), edge auto-scroll during drags,
21//! linked selection (click = link group, Alt+click = single, linked clips get a thin outline), audio
22//! volume line (dB mapped, draggable) + fade handles, video track V/S headers, full-width track resize
23//! handles, draggable keyframe diamonds with an easing context menu, transition bands (edge drag = duration,
24//! right-click = remove), colour labels / sequence tint, speed badges, thumbnail filmstrips (ThumbCache),
25//! auto-cut keep-range shading and Sequence open-on-double-click.
26//!
27//! Round 3: rubber-band select on empty lane space (Shift adds; dragging any selected clip moves the whole
28//! selection as one undo step), drag past the top video / bottom audio row to drop onto a freshly created
29//! track (gutter preview, Esc cancels the gesture), keyframe diamonds in a value lane on tall clips (y =
30//! value, x = time, tooltip, one undo per gesture), project markers on the ruler and clip markers inside
31//! clips (click selects, double-click seeks, drag moves, right-click = Rename / Delete / Set label),
32//! clip colours from `Project.labels`, hatched Adjustment layers with an "adj" badge, and the Add Marker /
33//! Copy & Paste Attributes / Add Mask / Nest / Convert to Adjustment Layer context-menu entries.

175pub fn paste_clips(p: &mut Project, clips: Vec<Clip>, assets: Vec<Asset>, at: f64, target: Option<usize>) -> Vec<Id> {
176    let ids = p.place_clips(clips, assets, at);
177    if let Some(ti) = target.filter(|&i| i < p.tracks.len()) {
178        let kind = p.tracks[ti].kind;
179        let list = if kind == TrackKind::Video { p.video_tracks() } else { p.audio_tracks() };
180        let row = |t: usize| list.iter().position(|&x| x == t);
181        let from = ids.iter().filter_map(|&id| p.track_of(id)).filter_map(row).min();
182        if let (Some(to), Some(from)) = (row(ti), from) {
183            p.move_clips(&ids, 0.0, to as i32 - from as i32, Some(kind));
184        }
185    }
186    ids
187}

189impl TimelineState {
190    /// Timeline time at an absolute x position (points).
191    pub fn time_at(&self, x: f32) -> f64 {
192        self.scroll_x + ((x - self.lanes_rect.left()) / self.zoom) as f64
193    }
194    pub fn x_at(&self, t: f64) -> f32 {
195        self.lanes_rect.left() + ((t - self.scroll_x) as f32) * self.zoom
196    }
197    /// Track index under an absolute y position (points), given the project (for track heights).
198    pub fn track_at(&self, y: f32, project: &Project) -> Option<usize> {
199        if y < self.lanes_rect.top() {
200            return None;
201        }
202        let mut top = self.lanes_rect.top() - self.scroll_y;
203        for i in row_order(project) {
204            let h = project.tracks[i].height;
205            if y >= top && y < top + h {
206                return Some(i);
207            }
208            top += h;
209        }
210        None
211    }
212    /// Fit [0, duration] into `avail_w` points.
213    pub fn zoom_to_fit(&mut self, duration: f64, avail_w: f32) {
214        let d = duration.max(1.0) as f32;
215        self.zoom = ((avail_w - 20.0).max(50.0) / d).clamp(0.5, 2000.0);
216        self.scroll_x = 0.0;
217    }
218    /// Multiply zoom, keeping the time under `anchor_x` (absolute points) fixed if given.
219    pub fn zoom_by(&mut self, factor: f32, anchor_x: Option<f32>) {
220        let anchor_t = anchor_x.map(|x| self.time_at(x));
221        self.zoom = (self.zoom * factor).clamp(0.5, 2000.0);
222        if let (Some(t), Some(x)) = (anchor_t, anchor_x) {
223            self.scroll_x = (t - ((x - self.lanes_rect.left()) / self.zoom) as f64).max(0.0);
224        }
225    }
226    /// Scroll so `t` is visible (used when the playhead runs off-screen while playing). Suspended after a
227    /// user pan until `t` is back in view.
228    /// Scroll `t` into view unconditionally (an explicit seek: Home/End, prev/next cut, a subtitle
229    /// click, an MCP `playback.seek`). Unlike `ensure_visible` a previous user pan does not suppress it.
230    pub fn follow_playhead(&mut self, t: f64) {
231        self.user_panned = false;
232        self.ensure_visible(t);
233    }
234    pub fn ensure_visible(&mut self, t: f64) {
235        let w = (self.lanes_rect.width() / self.zoom) as f64;
236        if w <= 0.0 {
237            return;
238        }
239        if t >= self.scroll_x && t <= self.scroll_x + w {
240            self.user_panned = false;
241        } else if !self.user_panned {
242            self.scroll_x = (t - w * 0.1).max(0.0);
243        }
244    }
245}

370/// Display order: video tracks reversed (Vn on top, V1 just above audio), then A1..An.
371fn row_order(p: &Project) -> impl Iterator<Item = usize> + '_ {
372    let n = p.tracks.len();
373    (0..n)
374        .rev()
375        .filter(move |&i| p.tracks[i].kind == TrackKind::Video)
376        .chain((0..n).filter(move |&i| p.tracks[i].kind == TrackKind::Audio))
377}

379/// Top y of track `ti` in display order (absolute points).
380fn row_top(state: &TimelineState, p: &Project, ti: usize) -> Option<f32> {
381    let mut top = state.lanes_rect.top() - state.scroll_y;
382    for i in row_order(p) {
383        if i == ti {
384            return Some(top);
385        }
386        top += p.tracks[i].height;
387    }
388    None
389}

391/// The clip under a drop at screen `pos` / timeline time `t`, with its lane rect — what an effect or a
392/// transition dragged out of its panel is aimed at (`app.rs` resolves the same clip from the reported
393/// time + track when the drop actually happens).
394fn drop_on_clip<'a>(state: &TimelineState, p: &'a Project, pos: Pos2, t: f64) -> Option<(Rect, &'a Clip)> {
395    let ti = state.track_at(pos.y, p)?;
396    let top = row_top(state, p, ti)?;
397    let clip = p.tracks.get(ti)?.clips.iter().find(|c| c.contains(t))?;
398    let rect = Rect::from_min_max(
399        pos2(state.x_at(clip.start), top + 1.0),
400        pos2(state.x_at(clip.end()), top + p.tracks[ti].height - 1.0),
401    );
402    Some((rect, clip))
403}

405/// Nearest candidate within `thr` of `t`.
406fn nearest(t: f64, thr: f64, candidates: impl Iterator<Item = f64>) -> Option<f64> {
407    let mut best: Option<f64> = None;
408    for x in candidates {
409        let d = (x - t).abs();
410        if d <= thr && best.map_or(true, |b| d < (b - t).abs()) {
411            best = Some(x);
412        }
413    }
414    best
415}

417/// Snap target for `t`: 0, playhead, in/out, edges of clips not in `exclude` — within `thr` seconds.
418fn snap_target(t: f64, thr: f64, p: &Project, playhead: f64, exclude: &[Id]) -> Option<f64> {
419    let edges = p.all_clips().filter(|(_, c)| !exclude.contains(&c.id)).flat_map(|(_, c)| [c.start, c.end()]);
420    nearest(t, thr, [0.0, playhead].into_iter().chain(p.in_point).chain(p.out_point).chain(edges))
421}

423/// Frame-quantise a pointer time and pull it onto the nearest snap candidate. Every tool that turns a
424/// pointer x into a time goes through here, so the razor, the marker tool, marker drags and media drops
425/// land on the same edges that moves and trims already snap to.
426/// Where the playhead lands when snapping is on: clip edges, in/out and 0, but NOT the playhead itself,
427/// which would always be the nearest candidate and pin it where it already is.
428fn snap_playhead(t: f64, on: bool, zoom: f32, p: &Project) -> f64 {
429    if !on {
430        return p.snap_frame(t);
431    }
432    let t = p.snap_frame(t);
433    let edges = p.all_clips().flat_map(|(_, c)| [c.start, c.end()]);
434    let cands = [0.0].into_iter().chain(p.in_point).chain(p.out_point).chain(edges);
435    nearest(t, (SNAP_PX / zoom) as f64, cands).unwrap_or(t)
436}

438fn snap_time(t: f64, on: bool, zoom: f32, p: &Project, playhead: f64, exclude: &[Id]) -> f64 {
439    if !on {
440        return t;
441    }
442    let t = p.snap_frame(t);
443    snap_target(t, (SNAP_PX / zoom) as f64, p, playhead, exclude).unwrap_or(t)
444}

446/// (major, minor) tick spacing in seconds for a zoom (px/s).
447fn tick_step(zoom: f32) -> (f64, f64) {
448    TICKS.iter().copied().find(|(major, _)| *major * zoom as f64 >= 80.0).unwrap_or(TICKS[TICKS.len() - 1])
449}

451fn tick_label(t: f64, major: f64) -> String {
452    let m = (t / 60.0).floor();
453    let s = t - m * 60.0;
454    if major >= 1.0 {
455        format!("{}:{:02}", m, s.round())
456    } else if major >= 0.1 {
457        format!("{}:{:04.1}", m, s)
458    } else {
459        format!("{}:{:05.2}", m, s)
460    }
461}

463fn toggle_button(
464    ui: &egui::Ui,
465    p: &egui::Painter,
466    rect: Rect,

469    on: bool,

481        }
482    }
483    r.clicked()
484}

486/// One vertical min..max line per point column over the visible part of an audio clip.
487fn draw_waveform(p: &egui::Painter, peaks: &Peaks, clip: &Clip, vis: Rect, state: &TimelineState, color: Color32) {
488    let mid = vis.center().y;
489    let half = (vis.height() * 0.5 - 1.0).max(0.0);
490    let stroke = Stroke::new(1.0, color);
491    let mut x = vis.left().floor();
492    while x < vis.right() {
493        let t0 = clip.src_time(state.time_at(x));
494        let t1 = clip.src_time(state.time_at(x + 1.0));
495        let (lo, hi) = peaks.range(t0, t1);
496        if hi > lo {
497            p.vline(x + 0.5, Rangef::new(mid - hi * half, mid - lo * half), stroke);
498        }
499        x += 1.0;
500    }
501}

524/// The range the rest of the UI enforces for property `i` (inspector sliders for the base props,
525/// `ParamSpec` for effect params), so a value-lane drag cannot write past what a `DragValue` allows.
526/// `None` where nothing enforces one: Position X/Y and Rotation are unbounded everywhere, and Volume is
527/// a gain behind a nonlinear dB slider.
528/// ponytail: same property order as curves.rs — move it next to `prop_ref` if a second caller shows up.
529fn prop_range(c: &Clip, i: usize) -> Option<(f64, f64)> {
530    let nb = if c.is_visual() { 6 } else { 3 };
531    if i < nb {
532        return match (c.is_visual(), i) {
533            (true, 2) => Some((0.01, 20.0)),              // Scale
534            (true, 4) => Some((0.0, 1.0)),                // Opacity
535            (false, 1) => Some((-1.0, 1.0)),              // Pan
536            (_, _) if i == nb - 1 => Some((0.01, 100.0)), // Speed — same clamp as Clip::set_speed
537            _ => None,
538        };
539    }
540    let mut i = i - nb;
541    for e in &c.effects {
542        if i < e.params.len() {
543            return e.specs().get(i).map(|s| (s.min, s.max));
544        }
545        i -= e.params.len();
546    }
547    None
548}

655/// dB → fraction of the clip height measured from the bottom. Piecewise linear: -60 dB = 0, 0 dB = 0.7,
656/// +12 dB = 1 (so unity gain sits at 70 % height and the usable attenuation range gets most of the clip).
657fn db_frac(db: f32) -> f32 {
658    if db >= 0.0 {
659        0.7 + (db / DB_TOP) * 0.3
660    } else {
661        0.7 * (1.0 - db / DB_BOT)
662    }
663}
664
665fn frac_db(f: f32) -> f32 {
666    if f >= 0.7 {
667        (f - 0.7) / 0.3 * DB_TOP
668    } else {
669        (1.0 - f / 0.7) * DB_BOT
670    }
671}
672
673fn gain_db(g: f32) -> f32 {
674    (20.0 * g.max(1e-4).log10()).clamp(DB_BOT, DB_TOP)
675}

677/// Filmstrip of source-time thumbnails across a video/image clip. Non-blocking: misses are queued in the
678/// ThumbCache worker and painted on a later frame.
679fn draw_filmstrip(
680    p: &egui::Painter,
681    ectx: &egui::Context,
682    state: &TimelineState,
683    clip: &Clip,
684    asset: &Asset,
685    rect: Rect,
686    vis: Rect,
687    thumbs: &mut ThumbCache,
688) {
689    let band = 16.0; // name row above the strip
690    let h = rect.height() - band;
691    if h < 12.0 || !vis.is_positive() {
692        return;
693    }
694    // ponytail: 16 px decode buckets, so a track-height drag reuses thumbs instead of queueing a fresh
695    // set (new cache key + new t grid) every frame. Ceiling: <5 % horizontal squash from the rounding.
696    let hq = (((h as u32 + 8) / 16) * 16).max(16);
697    let aspect = if asset.height > 0 { asset.width as f32 / asset.height as f32 } else { 16.0 / 9.0 };
698    let step = (hq as f32 * aspect).max(80.0);
699    let pc = p.with_clip_rect(vis);
700    let uv = Rect::from_min_max(pos2(0.0, 0.0), pos2(1.0, 1.0));
701    let mut n = ((vis.left() - rect.left()) / step).floor().max(0.0);
702    loop {
703        let x = rect.left() + n * step;
704        if x >= vis.right() || x >= rect.right() {
705            break;
706        }
707        let t = clip.src_time(state.time_at(x)).max(0.0);
708        if let Some((tex, [tw, th])) = thumbs.texture(ectx, &asset.path, t, hq) {
709            let w = h * tw as f32 / th.max(1) as f32;
710            pc.image(tex, Rect::from_min_size(pos2(x, rect.top() + band), vec2(w.min(step), h)), uv, Color32::WHITE);
711        }
712        n += 1.0;
713    }
714}

2500                if let Some(tr) = p.tracks.get_mut(*track) {
2501                    // duration follows the dragged edge: distance from the anchor (the cut, or the
2502                    // clip edge the transition hangs off), capped by what the clip(s) can supply
2503                    let lim = tr.transitions.iter().find(|t| t.id == *id).and_then(|t| {
2504                        let (l, r) = tr.transition_clips(t)?;
2505                        Some(match t.edge {
2506                            crate::model::TransitionEdge::Cut => (r?.start, 2.0, 2.0 * l?.duration.min(r?.duration)),
2507                            crate::model::TransitionEdge::In => (r?.start, 1.0, r?.duration),
2508                            crate::model::TransitionEdge::Out => (l?.end(), 1.0, l?.duration),
2509                        })
2510                    });
2511                    if let Some((anchor, scale, max)) = lim {
2512                        // cap: the Transitions panel's 0.1..5 s range, and what the clip(s) can actually supply
2513                        let d = ((t_at(pos.x) - anchor).abs() * scale).min(max).min(5.0).max(0.1);
2514                        if let Some(t) = tr.transitions.iter_mut().find(|t| t.id == *id) {
2515                            if (t.duration - d).abs() > 1e-9 {
2516                                t.duration = d;
2517                                *changed = true;
2518                            }
2519                        }
2520                    }
2521                }
2522            }
2523            Gesture::Marker { id, clip, changed } => {
2524                // markers always land on a frame; snapping additionally pulls them onto clip edges
2525                let want = p.snap_frame(t_at(pos.x).max(0.0));
2526                let want = if c.snap { snap_target(want, thr, p, *c.playhead, &[]).unwrap_or(want) } else { want };
2527                // clip markers are clip-local and stay inside their clip
2528                let nt = match clip.and_then(|cid| p.clip(cid)) {
2529                    Some(cl) => (want - cl.start).clamp(0.0, cl.duration),
2530                    None => want,
2531                };
2532                if let Some(m) = p.marker_mut(*id) {
2533                    if (m.t - nt).abs() > 1e-9 {
2534                        m.t = nt;
2535                        *changed = true;
2536                    }
2537                }
2538                if clip.is_none() && *changed {
2539                    p.sort_markers();
2540                }
2541            }
2542            Gesture::Spacer { ids, dt, room } => {
2543                // move_clips is all-or-nothing, so the clip in front of the group is never overrun
2544                let mut want = if c.snap { p.snap_frame(dx) } else { dx };
2545                if c.snap {
2546                    // what the user watches move is the group's leading edge: snap that, not the raw delta
2547                    let now = ids.iter().filter_map(|&id| p.clip(id)).map(|cl| cl.start).fold(f64::INFINITY, f64::min);
2548                    let lead = now - *dt; // where that edge sat before the gesture
2549                    if lead.is_finite() {
2550                        if let Some(t) = snap_target(lead + want, thr, p, *c.playhead, ids) {
2551                            want = t - lead;
2552                        }
2553                    }
2554                }
2555                let want = want.max(-*room);
2556                if (want - *dt).abs() > 1e-9 && p.move_clips(ids, want - *dt, 0, None) {
2557                    *dt = want;
2558                }
2559            }
2560        }
2561    }
2562    // ---- edge auto-scroll while dragging (playhead scrub, move/trim, keyframes, …) ----
2563    if state.drag.is_some() || scrub_x.is_some() {
2564        if let Some(pos) = pointer {
2565            let over = if pos.x < lanes.left() + SCROLL_MARGIN {
2566                pos.x - (lanes.left() + SCROLL_MARGIN)
2567            } else if pos.x > lanes.right() - SCROLL_MARGIN {
2568                pos.x - (lanes.right() - SCROLL_MARGIN)
2569            } else {
2570                0.0
2571            };
2572            if over != 0.0 {
2573                let ds = (over as f64 * 0.2) / state.zoom as f64; // ~proportional to the overshoot, per frame
2574                let ns = (state.scroll_x + ds).max(0.0);
2575                let applied = ns - state.scroll_x;
2576                if applied != 0.0 {
2577                    state.scroll_x = ns;
2578                    state.user_panned = true;
2579                    // keep pixel-based gestures (move/trim use pointer - origin) tracking the scrolled content
2580                    if let Some(d) = state.drag.as_mut() {
2581                        d.origin.x -= (applied * state.zoom as f64) as f32;
2582                    }
2583                    ui.ctx().request_repaint();
2584                }
2585            }
2586        }
2587    }
2588    if !primary_down {
2589        if let Some(d) = state.drag.take() {
2590            let mut edited = match &d.g {
2591                Gesture::Move { dt, dtrack, .. } => *dt != 0.0 || *dtrack != 0,
2592                Gesture::Spacer { dt, .. } => *dt != 0.0,
2593                Gesture::Trim { changed, .. }
2594                | Gesture::Stretch { changed, .. }
2595                | Gesture::Volume { changed, .. }
2596                | Gesture::Fade { changed, .. }
2597                | Gesture::Keys { changed, .. }
2598                | Gesture::TransDur { changed, .. }
2599                | Gesture::Marker { changed, .. } => *changed,
2600            };
2601            // released over the gutter: add a track at the far end and drop the clips of that kind on it
2602            if let Gesture::Move { ids, kind, new_track: true, .. } = &d.g {
2603                let p = &mut *c.project;
2604                let ti = p.add_track(*kind);
2605                let list = if *kind == TrackKind::Video { p.video_tracks() } else { p.audio_tracks() };
2606                let pos_of = |t: usize| list.iter().position(|&x| x == t).map(|i| i as i32);
2607                let from =
2608                    ids.iter().filter_map(|&cid| p.track_of(cid)).find(|&t| p.tracks[t].kind == *kind).and_then(pos_of);
2609                match from.zip(pos_of(ti)).filter(|(f, to)| f != to) {
2610                    Some((f, to)) if p.move_clips(ids, 0.0, to - f, Some(*kind)) => edited = true,
2611                    _ => p.remove_track(ti),
2612                }
2613            }
2614            if edited {
2615                (c.undo)(&d.before);
2616                out.edited = true;
2617            }
2618        }
2619    }
2620    out
2621}
