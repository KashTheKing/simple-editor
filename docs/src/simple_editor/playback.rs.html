1//! Player: owns the render thread (Compositor + DecoderPool → latest Frame) and the audio thread
2//! (Mixer + DecoderPool → ring buffer → cpal WASAPI output). The wall clock is the master when playing:
3//! time = base_time + elapsed. Video frames are rendered on the project fps grid (t quantized to
4//! frame index / fps, exactly like export) and published via `take_frame`; `ctx.request_repaint()`
5//! is called whenever a new frame is ready.
6//!
7//! Finished work is cached in RAM (`Cache`, ~512 MB LRU): composited frames on the CPU path, decoded
8//! layer sets on the GPU path, keyed by frame index. Replays and scrub-backs over recently seen
9//! footage are served without touching a decoder. Any edit (SetProject), canvas resize, backend or
10//! GPU-mode switch clears the cache — entries can never go stale. While playing, the pacing gap
11//! until the next frame is due is spent pre-rendering upcoming frames into the cache instead of
12//! sleeping, so a decode hiccup lands in the prefetch window and not on a visible frame.
13//!
14//! Commands (mpsc from the UI thread): SetProject, Seek, Play, Pause, Canvas, Backend, ClearDecoders(ack),
15//! Quit. The payload (project, clock, canvas) lives in `Shared`, so draining the queue and reading the
16//! latest state gives "latest request wins" for free. While paused, every Seek/SetProject/Canvas
17//! re-renders the current time; the render thread blocks on the channel when idle (zero CPU).
18//! Audio thread mirrors the clock: when playing it keeps ~120 ms of mixed audio ahead in the ring;
19//! seek/pause flush the ring. Reaching the project end pauses (the UI sees is_playing() flip).
20//!
21//! GPU mode (`set_gpu(true)`, settings.gpu + a working `engine::gpu::GpuRenderer`): the render thread
22//! stops compositing and instead publishes the **layers** for the current instant (`take_layers` →
23//! `engine::gpu::LayerSet`); the UI thread renders them with the GL context it owns.
24//! Everything else (clock, pacing, decoder pool, audio) is identical, so falling back to the CPU
25//! compositor is a single `set_gpu(false)`.

40/// Audio mixed per block (frames) and how far ahead of the clock the ring is kept.
41const BLOCK: usize = 1024;
42const LEAD_SECS: f64 = 0.12;
43/// RAM the render thread's finished-work cache may hold.
44/// ponytail: fixed 512 MB — a settings knob (or scaling to installed RAM) is the upgrade if it matters.
45const CACHE_BYTES: usize = 512 << 20;
46/// How far past the playhead the prefetcher keeps frames ready (seconds).
47const READ_AHEAD_SECS: f64 = 1.5;
48/// Recently shown frames kept eviction-protected behind the playhead (seconds), so a short
49/// scrub-back replays from the cache.
50const TRAIL_SECS: f64 = 0.5;
51/// How far the clock may run past the newest published frame before playback is declared
52/// buffering (the UI pauses the clock and shows a spinner until the cache refills).
53const STALL_BEHIND: f64 = 0.15;
54/// Recycled frame buffers kept for the CPU compositor (cache evictions feed it).
55const FPOOL_KEEP: usize = 4;

57/// Byte-budgeted LRU keyed by frame index on the project fps grid. The whole cache is cleared on
58/// any change that could affect the picture, so entries never go stale.
59struct Cache<T> {
60    /// frame index -> (last-use tick, payload bytes, payload)
61    map: HashMap<i64, (u64, usize, Arc<T>)>,
62    bytes: usize,
63    tick: u64,
64    /// Inclusive index range evicted last (playhead trail + prefetch horizon).
65    protect: (i64, i64),
66}
67
68impl<T> Cache<T> {
69    fn new() -> Self {
70        Cache { map: HashMap::new(), bytes: 0, tick: 0, protect: (i64::MAX, i64::MIN) }
71    }
72    /// Frames in [lo, hi] are evicted only when nothing else is left to evict.
73    fn set_protect(&mut self, lo: i64, hi: i64) {
74        self.protect = (lo, hi);
75    }
76    fn get(&mut self, idx: i64) -> Option<Arc<T>> {
77        self.tick += 1;
78        let (t, _, v) = self.map.get_mut(&idx)?;
79        *t = self.tick;
80        Some(v.clone())
81    }
82    fn contains(&self, idx: i64) -> bool {
83        self.map.contains_key(&idx)
84    }
85    /// Insert and evict LRU entries down to the budget; evicted payloads come back for buffer recycling.
86    fn insert(&mut self, idx: i64, bytes: usize, v: Arc<T>) -> Vec<Arc<T>> {
87        self.tick += 1;
88        let mut out = Vec::new();
89        if let Some((_, b, old)) = self.map.insert(idx, (self.tick, bytes, v)) {
90            self.bytes -= b;
91            out.push(old);
92        }
93        self.bytes += bytes;
94        while self.bytes > CACHE_BYTES {
95            // ponytail: O(n) min scan per eviction; n is a few hundred at most. A heap if it ever shows up.
96            let (lo, hi) = self.protect;
97            let pick = self
98                .map
99                .iter()
100                .filter(|(i, _)| !(lo..=hi).contains(i))
101                .min_by_key(|(_, (t, _, _))| *t)
102                .or_else(|| self.map.iter().min_by_key(|(_, (t, _, _))| *t));
103            let Some((&i, _)) = pick else { break };
104            let (_, b, old) = self.map.remove(&i).expect("key from iter");
105            self.bytes -= b;
106            out.push(old);
107        }
108        out
109    }
110    /// Drop every entry whose index lies in [lo, hi], handing payloads back for recycling.
111    fn evict_range(&mut self, lo: i64, hi: i64) -> Vec<Arc<T>> {
112        let keys: Vec<i64> = self.map.keys().copied().filter(|i| (lo..=hi).contains(i)).collect();
113        let mut out = Vec::new();
114        for i in keys {
115            let (_, b, v) = self.map.remove(&i).expect("key from keys");
116            self.bytes -= b;
117            out.push(v);
118        }
119        out
120    }
121    /// Empty the cache, handing every payload back for buffer recycling.
122    fn drain(&mut self) -> Vec<Arc<T>> {
123        self.bytes = 0;
124        self.map.drain().map(|(_, (_, _, v))| v).collect()
125    }
126}

128/// Wall clock + canvas shared by the UI and both threads. `now()` flips `playing` off at the end.
129struct Clock {
130    playing: bool,
131    base_t: f64,
132    base_at: Instant,
133    duration: f64,
134    /// Render size in pixels (already clamped to the preview max width); (0,0) = nothing to render.
135    canvas: (u32, u32),
136}
137
138impl Clock {
139    fn now(&mut self) -> f64 {
140        if !self.playing {
141            return self.base_t;
142        }
143        let t = self.base_t + self.base_at.elapsed().as_secs_f64();
144        if t < self.duration {
145            return t;
146        }
147        self.playing = false;
148        self.base_t = self.duration;
149        self.duration
150    }
151}

335    /// Synchronously drop every decoder on both threads (before overwriting a source file).
336    pub fn release_files(&mut self) {
337        let (tx, rx) = mpsc::sync_channel(2);
338        self.both(Cmd::ClearDecoders(tx));
339        let deadline = Instant::now() + Duration::from_secs(2);
340        for _ in 0..2 {
341            // a dead thread drops its sender → Disconnected → we stop waiting early
342            if rx.recv_timeout(deadline.saturating_duration_since(Instant::now())).is_err() {
343                break;
344            }
345        }
346    }

355/// Run one decode+compose/mix under `catch_unwind`: a panicking decoder must not kill the render or audio
356/// thread for the rest of the session (the UI would freeze on the last frame with no error). On a panic the
357/// decoders are dropped so the next call starts clean. False = the call panicked and its output is garbage.
358fn guarded(pool: &mut DecoderPool, f: impl FnOnce(&mut DecoderPool)) -> bool {
359    if std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| f(pool))).is_ok() {
360        return true;
361    }
362    pool.clear();
363    false
364}

366fn render_thread(
367    shared: Arc<Shared>,
368    rx: Receiver<Cmd>,
369    ctx: eframe::egui::Context,
370    backend: Backend,
371    text: Arc<Mutex<TextRasterizer>>,
372) {
373    let mut pool = DecoderPool::new(backend);
374    let mut comp = Compositor::new();
375    let mut shapes = ShapeRasterizer::new();
376    let mut project = lock(&shared.project).clone();
377    let mut dirty = false;
378    let mut pending: Option<Cmd> = None;
379    // GPU mode state + recycled layer buffers
380    let mut gpu = false;
381    let mut spare_layers: Vec<Frame> = Vec::new();
382    // finished-work caches, keyed by frame index on the fps grid (see module docs), plus a small pool
383    // of recycled Frame buffers for the CPU compositor (fed by cache evictions)
384    let mut fcache: Cache<Frame> = Cache::new();
385    let mut lcache: Cache<LayerSet> = Cache::new();
386    let mut fpool: Vec<Frame> = Vec::new();
387    let mut last_pub: i64 = -1;
388    // buffering: the clock outran decode; free-run the prefetcher (even while the UI pauses the
389    // clock) until the read-ahead is back, instead of live-rendering straight into every stall
390    let mut stall = false;
391    // clear both caches, recycling their buffers (`drain` so a per-edit clear does not thrash the allocator)
392    macro_rules! clear_caches {
393        () => {
394            for old in fcache.drain() {
395                reclaim(old, &mut fpool);
396            }
397            for old in lcache.drain() {
398                recycle(Some(old), &mut spare_layers);
399            }
400            last_pub = -1;
401        };
402    }
403    loop {
404        let playing = {
405            let mut c = lock(&shared.clock);
406            c.now();
407            c.playing
408        };
409        if pending.is_none() && !dirty && !playing && !stall {
410            let Ok(c) = rx.recv() else { return }; // idle: block, zero CPU
411            pending = Some(c);
412        }
413        // drain everything queued; the shared state already holds the latest values
414        loop {
415            let c = match pending.take() {
416                Some(c) => c,
417                None => match rx.try_recv() {
418                    Ok(c) => c,
419                    Err(TryRecvError::Empty) => break,
420                    Err(TryRecvError::Disconnected) => return,
421                },
422            };
423            match c {
424                Cmd::SetProject => {
425                    let new = lock(&shared.project).clone();
426                    dirty = true;
427                    // evict only the frames the edit can have changed; None = anything could differ
428                    match video_dirty_spans(&project, &new) {
429                        Some(spans) => {
430                            let fps = new.fps.max(1.0);
431                            for (a, b) in spans {
432                                let (lo, hi) = ((a * fps).floor() as i64 - 1, (b * fps).ceil() as i64 + 1);
433                                for old in fcache.evict_range(lo, hi) {
434                                    reclaim(old, &mut fpool);
435                                }
436                                for old in lcache.evict_range(lo, hi) {
437                                    recycle(Some(old), &mut spare_layers);
438                                }
439                            }
440                            last_pub = -1; // the current frame may sit in an evicted span: republish
441                        }
442                        None => {
443                            clear_caches!();
444                        }
445                    }
446                    project = new;
447                }
448                Cmd::Seek | Cmd::Play | Cmd::Pause => dirty = true,
449                Cmd::Canvas => {
450                    dirty = true;
451                    clear_caches!(); // cached entries are the old canvas size
452                }
453                Cmd::Backend(b) => {
454                    pool.set_backend(b);
455                    dirty = true;
456                    clear_caches!();
457                }
458                Cmd::Gpu(on) => {
459                    if gpu != on {
460                        gpu = on;
461                        *lock(&shared.layers) = None;
462                        dirty = true;
463                        clear_caches!();
464                    }
465                }
466                Cmd::Proxies(map) => {
467                    pool.set_proxies(map);
468                    dirty = true;
469                    clear_caches!(); // cached frames may show the other source now
470                }
471                Cmd::ClearDecoders(ack) => {
472                    pool.clear();
473                    clear_caches!(); // the caller is about to overwrite a source file
474                    let _ = ack.send(());
475                }
476                Cmd::LayersOnce(t, max_w, reply) => {
477                    let (pw, ph) = (project.width.max(1), project.height.max(1));
478                    let w = pw.min(max_w.max(16));
479                    let h = ((ph as u64 * w as u64) / pw as u64).max(1) as u32;
480                    let mut set = LayerSet::default();
481                    let text = &mut lock(&text);
482                    guarded(&mut pool, |pool| {
483                        set = decode_layers(&project, t, w, h, pool, &mut spare_layers, text, &mut shapes, &mut comp)
484                    });
485                    let _ = reply.send(Arc::new(set));
486                }
487                Cmd::RenderOnce(t, max_w, reply) => {
488                    let (pw, ph) = (project.width.max(1), project.height.max(1));
489                    let w = pw.min(max_w.max(16));
490                    let h = ((ph as u64 * w as u64) / pw as u64).max(1) as u32;
491                    let mut f = Frame::default();
492                    guarded(&mut pool, |pool| comp.render(&project, t, w, h, pool, &mut lock(&text), &mut f));
493                    let _ = reply.send(Arc::new(f));
494                }
495                Cmd::Quit => return,
496            }
497        }
498        let (playing, t, (w, h), duration) = {
499            let mut c = lock(&shared.clock);
500            let t = c.now();
501            (c.playing, t, c.canvas, c.duration)
502        };
503        if !playing && !dirty && !stall {
504            continue;
505        }
506        let force = std::mem::take(&mut dirty);
507        let fps = project.fps.max(1.0);
508        if w == 0 || h == 0 {
509            stall = false;
510            shared.buffering.store(false, Ordering::Relaxed);
511            if playing {
512                std::thread::sleep(Duration::from_secs_f64(1.0 / fps)); // nothing to render, keep pace
513            }
514            continue;
515        }
516        // quantize to the fps grid (same formula as ffpipe): one render per frame index, cache-keyable
517        let idx = (t * fps + 1e-6).floor() as i64;
518        // prefetch horizon: READ_AHEAD_SECS of frames, capped so the window can't blow the cache
519        // budget on its own (GPU layer sets are conservatively costed like one frame per index)
520        let frame_bytes = (w as usize * h as usize * 4).max(1);
521        let read_ahead =
522            ((fps * READ_AHEAD_SECS).ceil() as i64).clamp(8, (CACHE_BYTES / 2 / frame_bytes).max(8) as i64);
523        let last_idx = (((duration * fps).ceil() as i64) - 1).max(idx);
524        // keep the trail + horizon around the playhead eviction-protected: a scrub-back replays free
525        let trail = (fps * TRAIL_SECS).ceil() as i64;
526        fcache.set_protect(idx - trail, idx + read_ahead);
527        lcache.set_protect(idx - trail, idx + read_ahead);
528        macro_rules! cached {
529            ($i:expr) => {
530                if gpu {
531                    lcache.contains($i)
532                } else {
533                    fcache.contains($i)
534                }
535            };
536        }
537        // buffering: the clock ran STALL_BEHIND past the newest published frame and the due frame
538        // still isn't cached — declare a stall instead of grinding out one late frame at a time.
539        // The UI polls is_buffering(), pauses the clock (audio flushes with it) and shows a spinner.
540        if playing && !stall && last_pub >= 0 && idx > last_pub {
541            let behind = t - (last_pub + 1) as f64 / fps;
542            if behind > STALL_BEHIND && !cached!(idx) {
543                stall = true;
544                shared.buffering.store(true, Ordering::Relaxed);
545                ctx.request_repaint();
546            }
547        }
548        if stall {
549            // free-run the prefetcher: one frame per loop pass so commands stay responsive
550            let horizon = (idx + read_ahead).min(last_idx);
551            if let Some(i) = (idx..=horizon).find(|i| !cached!(*i)) {
552                let ok = if gpu {
553                    gpu_cached(
554                        &mut lcache,
555                        &mut spare_layers,
556                        &project,
557                        i,
558                        fps,
559                        w,
560                        h,
561                        &mut pool,
562                        &text,
563                        &mut shapes,
564                        &mut comp,
565                        &shared.cache_hits,
566                    )
567                    .1
568                } else {
569                    cpu_cached(
570                        &mut fcache,
571                        &mut fpool,
572                        &project,
573                        i,
574                        fps,
575                        w,
576                        h,
577                        &mut pool,
578                        &text,
579                        &mut comp,
580                        &shared.cache_hits,
581                    )
582                    .1
583                };
584                if !ok {
585                    stall = false; // a decode panicked and the pool was cleared: give up on this stall
586                    shared.buffering.store(false, Ordering::Relaxed);
587                    ctx.request_repaint();
588                }
589            }
590            // enough contiguous frames ready (hysteresis: a third of the horizon) -> resume
591            let goal = (idx + (read_ahead / 3).max(2)).min(last_idx);
592            if stall && (idx..=goal).all(|i| cached!(i)) {
593                stall = false;
594                shared.buffering.store(false, Ordering::Relaxed);
595                dirty = true; // republish the (now cached) due frame on the next pass
596                ctx.request_repaint();
597            }
598            continue; // re-drain commands between decodes; never live-render while buffering
599        }
600        if force || idx != last_pub {
601            if gpu {
602                // the UI thread owns the GL context: decode/rasterise here, render there
603                let (set, _) = gpu_cached(
604                    &mut lcache,
605                    &mut spare_layers,
606                    &project,
607                    idx,
608                    fps,
609                    w,
610                    h,
611                    &mut pool,
612                    &text,
613                    &mut shapes,
614                    &mut comp,
615                    &shared.cache_hits,
616                );
617                *lock(&shared.layers) = Some(set);
618            } else {
619                let (frame, _) = cpu_cached(
620                    &mut fcache,
621                    &mut fpool,
622                    &project,
623                    idx,
624                    fps,
625                    w,
626                    h,
627                    &mut pool,
628                    &text,
629                    &mut comp,
630                    &shared.cache_hits,
631                );
632                *lock(&shared.frame) = Some(frame); // replaces an untaken (stale) frame: latest wins
633            }
634            last_pub = idx;
635            ctx.request_repaint();
636        }
637        if playing {
638            // Pacing window until frame idx+1 is due on the grid: pre-render upcoming frames into the
639            // cache, then sleep out whatever is left. A slow prefetch can overrun the window by one
640            // render — the same stall the live path would have hit at that frame's due time, just earlier.
641            // If we are behind, remain <= 0 and we render the current clock frame next (never queue up).
642            // ponytail: thread::sleep uses Windows' high-resolution waitable timer; recv_timeout rounds to
643            // the 15.6 ms scheduler tick (30 fps -> ~21 fps). timeBeginPeriod(1) would also work but costs power.
644            loop {
645                let remain = (idx + 1) as f64 / fps - lock(&shared.clock).now();
646                if remain <= 0.0 {
647                    break;
648                }
649                let horizon = (((duration * fps).ceil() as i64) - 1).min(idx + read_ahead);
650                let missing =
651                    ((idx + 1)..=horizon).find(|i| if gpu { !lcache.contains(*i) } else { !fcache.contains(*i) });
652                let Some(i) = missing else {
653                    std::thread::sleep(Duration::from_secs_f64(remain));
654                    break;
655                };
656                let ok = if gpu {
657                    gpu_cached(
658                        &mut lcache,
659                        &mut spare_layers,
660                        &project,
661                        i,
662                        fps,
663                        w,
664                        h,
665                        &mut pool,
666                        &text,
667                        &mut shapes,
668                        &mut comp,
669                        &shared.cache_hits,
670                    )
671                    .1
672                } else {
673                    cpu_cached(
674                        &mut fcache,
675                        &mut fpool,
676                        &project,
677                        i,
678                        fps,
679                        w,
680                        h,
681                        &mut pool,
682                        &text,
683                        &mut comp,
684                        &shared.cache_hits,
685                    )
686                    .1
687                };
688                if !ok {
689                    break; // a decode panicked and the pool was cleared: stop prefetching this window
690                }
691            }
692        }
693    }
694}

696/// Timeline spans (seconds) whose cached frames the edit `old` -> `new` can have changed.
697/// None = the change can affect any frame (or is too entangled to bound): clear everything.
698/// Edits that cannot change pixels (markers, in/out points, audio tracks, buses, planner, notes,
699/// labels, folders) produce no spans at all — the whole cache survives them.
700fn video_dirty_spans(old: &Project, new: &Project) -> Option<Vec<(f64, f64)>> {
701    // serde_json string equality instead of a PartialEq derive cascade across the whole model
702    fn json<T: serde::Serialize>(v: &T) -> String {
703        serde_json::to_string(v).unwrap_or_default()
704    }
705    // trivially frame-global state: any difference clears everything
706    if old.width != new.width
707        || old.height != new.height
708        || old.fps != new.fps
709        || old.editing != new.editing
710        || json(&old.scaler) != json(&new.scaler)
711        || old.show_subtitles != new.show_subtitles
712        || old.tracks.len() != new.tracks.len()
713        || json(&old.assets) != json(&new.assets)
714        || json(&old.sequences) != json(&new.sequences)
715    {
716        return None;
717    }
718    let mut spans: Vec<(f64, f64)> = Vec::new();
719    let clip_span = |c: &Clip| (c.start, c.start + c.duration.max(0.0));
720    // subtitle style/margin or cue edits dirty every cue window from both sides (cues are few)
721    if json(&old.subtitle_style) != json(&new.subtitle_style)
722        || old.subtitle_margin != new.subtitle_margin
723        || json(&old.subtitles) != json(&new.subtitles)
724    {
725        for c in old.subtitles.iter().chain(&new.subtitles) {
726            spans.push((c.start, c.end));
727        }
728    }
729    for (ot, nt) in old.tracks.iter().zip(&new.tracks) {
730        if ot.kind != nt.kind {
731            return None;
732        }
733        if ot.kind != TrackKind::Video {
734            continue; // audio-only edits never touch a rendered frame
735        }
736        // mute/solo flips change visibility rules across tracks: not worth bounding
737        if ot.muted != nt.muted || ot.solo != nt.solo {
738            return None;
739        }
740        // a transition edit repaints its window; the exact window needs the clip pair, so take the
741        // track's whole span (transition edits are rare; clip edits below stay tightly bounded)
742        if json(&ot.transitions) != json(&nt.transitions) {
743            spans.push((0.0, ot.end().max(nt.end())));
744            continue;
745        }
746        let olds: HashMap<crate::model::Id, &Clip> = ot.clips.iter().map(|c| (c.id, c)).collect();
747        let news: HashMap<crate::model::Id, &Clip> = nt.clips.iter().map(|c| (c.id, c)).collect();
748        for c in &ot.clips {
749            match news.get(&c.id) {
750                None => spans.push(clip_span(c)), // removed
751                Some(n) => {
752                    if json(*n) != json(c) {
753                        spans.push(clip_span(c));
754                        spans.push(clip_span(n));
755                    }
756                }
757            }
758        }
759        for c in nt.clips.iter().filter(|c| !olds.contains_key(&c.id)) {
760            spans.push(clip_span(c)); // added
761        }
762    }
763    Some(spans)
764}

766/// Take a frame buffer nobody uses any more back into the compositor's pool.
767fn reclaim(old: Arc<Frame>, fpool: &mut Vec<Frame>) {
768    if fpool.len() < FPOOL_KEEP {
769        if let Ok(f) = Arc::try_unwrap(old) {
770            fpool.push(f);
771        }
772    }

775/// Composited frame for grid index `idx` — from the cache when possible, else rendered and cached.
776/// `false` = the render panicked (nothing was cached; the returned frame is stale/blank).
777#[allow(clippy::too_many_arguments)]
778fn cpu_cached(
779    cache: &mut Cache<Frame>,
780    fpool: &mut Vec<Frame>,
781    project: &Project,
782    idx: i64,
783    fps: f64,
784    w: u32,
785    h: u32,
786    pool: &mut DecoderPool,
787    text: &Mutex<TextRasterizer>,
788    comp: &mut Compositor,
789    hits: &AtomicU64,
790) -> (Arc<Frame>, bool) {
791    if let Some(f) = cache.get(idx) {
792        hits.fetch_add(1, Ordering::Relaxed);
793        return (f, true);
794    }
795    let t = idx as f64 / fps;
796    let mut frame = fpool.pop().unwrap_or_default();
797    let ok = guarded(pool, |pool| comp.render(project, t, w, h, pool, &mut lock(text), &mut frame));
798    let frame = Arc::new(frame);
799    if ok {
800        for old in cache.insert(idx, frame.rgba.len(), frame.clone()) {
801            reclaim(old, fpool);
802        }
803    }
804    (frame, ok)
805}

807/// Decoded layer set for grid index `idx` — from the cache when possible, else decoded and cached.
808#[allow(clippy::too_many_arguments)]
809fn gpu_cached(
810    cache: &mut Cache<LayerSet>,
811    spare: &mut Vec<Frame>,
812    project: &Project,
813    idx: i64,
814    fps: f64,
815    w: u32,
816    h: u32,
817    pool: &mut DecoderPool,
818    text: &Mutex<TextRasterizer>,
819    shapes: &mut ShapeRasterizer,
820    comp: &mut Compositor,
821    hits: &AtomicU64,
822) -> (Arc<LayerSet>, bool) {
823    if let Some(s) = cache.get(idx) {
824        hits.fetch_add(1, Ordering::Relaxed);
825        return (s, true);
826    }
827    let t = idx as f64 / fps;
828    let mut set = LayerSet::default();
829    let ok = {
830        let text = &mut lock(text);
831        guarded(pool, |pool| set = decode_layers(project, t, w, h, pool, spare, text, shapes, comp))
832    };
833    let set = Arc::new(set);
834    if ok {
835        let bytes: usize = set.layers.iter().map(|(_, f)| f.rgba.len()).sum::<usize>()
836            + set.motion.iter().map(|(_, _, f)| f.rgba.len()).sum::<usize>();
837        for old in cache.insert(idx, bytes, set.clone()) {
838            recycle(Some(old), spare);
839        }
840    }
841    (set, ok)
842}

844/// Everything `engine::gpu` needs for one timeline instant (its `LayerSet` contract): one bitmap per
845/// visual clip on screen at `t` — **both** clips of a transition, which the renderer draws virtually
846/// extended past their own bounds — plus the subtitle bitmap under `LayerSet::SUBTITLES`. Media is
847/// decoded at the size the compositor would have used (placement, capped at the source's native size);
848/// text/shape clips are rasterised and nested sequences composited on the CPU, all at (w, h) — which is
849/// already the quality-scaled render size, so `gpu::render_size` must not be applied again. Adjustment
850/// layers need no bitmap (they re-process the canvas). Effects that `needs_motion()` also get neighbours.
851#[allow(clippy::too_many_arguments)]
852pub(crate) fn decode_layers(
853    project: &Project,
854    t: f64,
855    w: u32,
856    h: u32,
857    pool: &mut DecoderPool,
858    spare: &mut Vec<Frame>,
859    text: &mut TextRasterizer,
860    shapes: &mut ShapeRasterizer,
861    comp: &mut Compositor,
862) -> LayerSet {
863    let mut set = LayerSet::default();
864    for (ti, track) in project.tracks.iter().enumerate() {
865        if track.kind != TrackKind::Video || !project.active(ti) {
866            continue;
867        }
868        // mirrors gpu::render_canvas / compose::render_tracks: inside a transition window both clips are
869        // drawn past their own bounds, so filtering by `contains(t)` here would hard-cut every transition
870        if let Some((_, left, right)) = track.transition_at(t) {
871            for clip in [left, right].into_iter().flatten() {
872                layer_for(project, clip, t, w, h, pool, spare, text, shapes, comp, &mut set);
873            }
874            continue;
875        }
876        for clip in &track.clips {
877            if clip.enabled && clip.contains(t) {
878                layer_for(project, clip, t, w, h, pool, spare, text, shapes, comp, &mut set);
879            }
880        }
881    }
882    if project.show_subtitles {
883        if let Some(cue) = project.cue_at(t).filter(|c| !c.text.trim().is_empty()) {
884            let mut style = project.subtitle_style.clone();
885            style.text.clone_from(&cue.text);
886            let img = text.render(&style, w as f32 / project.width.max(1) as f32);
887            if img.width > 1 || img.height > 1 {
888                set.layers.push((LayerSet::SUBTITLES, img));
889            }
890        }
891    }
892    set
893}

895/// One clip's layer: video/images decode, text/shapes rasterise, a nested sequence is composited on the
896/// CPU. Kinds the renderer draws without a bitmap (Adjustment) or not at all (Audio) push nothing.
897#[allow(clippy::too_many_arguments)]
898fn layer_for(
899    project: &Project,
900    clip: &Clip,
901    t: f64,
902    w: u32,
903    h: u32,
904    pool: &mut DecoderPool,
905    spare: &mut Vec<Frame>,
906    text: &mut TextRasterizer,
907    shapes: &mut ShapeRasterizer,
908    comp: &mut Compositor,
909    set: &mut LayerSet,
910) {
911    // Asset nodes sample footage that need not be on the timeline, so nothing else decodes it: one
912    // frame per asset, keyed by the asset id (`gpu::eval_graph_on` looks it up there). Before the
913    // `draws()` gate, since an adjustment layer can hold them too.
914    // ponytail: sampled at the clip's own local time, at canvas size — no per-node time offset yet.
915    for aid in clip.graph.iter().flat_map(|g| g.nodes.iter()).filter_map(|n| match n.kind {
916        crate::model::NodeKind::Asset(a) => Some(a),
917        _ => None,
918    }) {
919        let Some(asset) = project.asset(aid).filter(|a| a.width > 0 && a.height > 0) else { continue };
920        if set.get(aid).is_some() {
921            continue;
922        }
923        let (dw, dh) = (w.min(asset.width), h.min(asset.height));
924        if let Some(f) = decode_one(pool, &asset.path, clip.local(t), asset.duration, dw, dh, spare) {
925            set.layers.push((aid, f));
926        }
927    }
928    if !clip.draws() {
929        return;
930    }
931    let s = w as f32 / project.width.max(1) as f32;
932    match clip.kind {
933        ClipKind::Video | ClipKind::Image => {
934            if clip.is_empty_container() {
935                let label_text = if !clip.container_label.is_empty() {
936                    format!("Container Slot\n[{}]", clip.container_label)
937                } else {
938                    "Container Slot\n(Empty)".to_string()
939                };
940                let style = crate::model::TextStyle {
941                    text: label_text,
942                    size: 32.0,
943                    color: [220, 220, 220, 255],
944                    box_color: [40, 40, 40, 220],
945                    box_padding: 16.0,
946                    align: 1,
947                    ..Default::default()
948                };
949                let img = text.render(&style, s);
950                if img.width > 1 || img.height > 1 {
951                    set.layers.push((clip.id, img));
952                }
953                return;
954            }
955            let Some(asset) = project.asset(clip.asset) else { return };
956            let native = (asset.width, asset.height);
957            if native.0 == 0 || native.1 == 0 {
958                return;
959            }
960            let p = placement(project, clip, t, native, w, h, true);
961            if !(p.w.is_finite() && p.h.is_finite()) {
962                return;
963            }
964            let dw = (p.w.round().max(1.0) as u32).clamp(1, native.0);
965            let dh = (p.h.round().max(1.0) as u32).clamp(1, native.1);
966            if let Some(f) = decode_one(pool, &asset.path, clip.src_time(t), asset.duration, dw, dh, spare) {
967                set.layers.push((clip.id, f));
968            }
969            // ponytail: two extra samples half a frame apart feed shutter-based effects; a shutter-angle
970            // driven sample count is the upgrade if motion blur ever needs to be exact.
971            if clip.effects.iter().any(|e| e.enabled && e.kind.needs_motion()) {
972                let fd = project.frame_dur();
973                for k in [-1.0, -0.5] {
974                    let st = clip.src_time(t + k * fd);
975                    if let Some(f) = decode_one(pool, &asset.path, st, asset.duration, dw, dh, spare) {
976                        set.motion.push((clip.id, k * fd, f));
977                    }
978                }
979            }
980        }
981        ClipKind::Text => {
982            let Some(style) = &clip.text else { return };
983            let img = text.render(style, s);
984            if img.width > 1 || img.height > 1 {
985                set.layers.push((clip.id, img));
986            }
987        }
988        ClipKind::Shape => {
989            let Some(style) = &clip.shape else { return };
990            let img = shapes.render(style, s, clip.local(t));
991            if !img.is_empty() {
992                set.layers.push((clip.id, img));
993            }
994        }
995        ClipKind::Sequence => {
996            let editing = project.editing == Some(clip.sequence);
997            let (qw, qh) = if editing {
998                (project.width, project.height) // its live size is swapped into the project
999            } else {
1000                match project.sequence(clip.sequence) {
1001                    Some(sq) => (sq.width, sq.height),
1002                    None => return,
1003                }
1004            };
1005            if qw == 0 || qh == 0 {
1006                return;
1007            }
1008            let p = placement(project, clip, t, (qw, qh), w, h, true);
1009            if !(p.w.is_finite() && p.h.is_finite() && p.w > 0.0 && p.h > 0.0) {
1010                return;
1011            }
1012            let dw = (p.w.round() as u32).clamp(1, qw);
1013            let dh = (p.h.round() as u32).clamp(1, qh);
1014            let dur = project.sequence_duration(clip.sequence);
1015            let st = clip.src_time(t).clamp(0.0, (dur - 1e-6).max(0.0));
1016            // ponytail: gpu.rs wants a nested sequence as a ready bitmap, so the CPU compositor renders
1017            // it — through a shallow project view (one Project clone per frame while a sequence clip is
1018            // on screen). Making `Compositor::render_tracks` pub would drop both the clone and this.
1019            let mut sub = project.clone();
1020            if !editing {
1021                let Some(sq) = project.sequence(clip.sequence) else { return };
1022                sub.tracks.clone_from(&sq.tracks);
1023                (sub.width, sub.height) = (qw, qh);
1024                sub.editing = Some(clip.sequence);
1025            }
1026            sub.show_subtitles = false; // subtitles belong to the outer canvas
1027            let mut frame = spare.pop().unwrap_or_default();
1028            comp.render(&sub, st, dw, dh, pool, text, &mut frame);
1029            set.layers.push((clip.id, Arc::new(frame)));
1030        }
1031        ClipKind::Audio | ClipKind::Adjustment => {}
1032    }
1033}

1035/// One decode into a recycled buffer. None when the decoder or the frame is unavailable.
1036fn decode_one(
1037    pool: &mut DecoderPool,
1038    path: &str,
1039    src_t: f64,
1040    src_duration: f64,
1041    dw: u32,
1042    dh: u32,
1043    spare: &mut Vec<Frame>,
1044) -> Option<Arc<Frame>> {
1045    let st = if src_duration > 0.0 { src_t.clamp(0.0, (src_duration - 1e-4).max(0.0)) } else { src_t.max(0.0) };
1046    let mut frame = spare.pop().unwrap_or_default();
1047    let dec = pool.video(path)?;
1048    if !dec.frame_at(st, dw, dh, &mut frame) {
1049        spare.push(frame);
1050        return None;
1051    }
1052    Some(Arc::new(frame))
1053}

1055/// Take the buffers of a layer set nobody uses any more back into the spare pool.
1056fn recycle(old: Option<Arc<LayerSet>>, spare: &mut Vec<Frame>) {
1057    const KEEP: usize = 8;
1058    let Some(Ok(set)) = old.map(Arc::try_unwrap) else { return };
1059    let frames = set.layers.into_iter().map(|(_, f)| f).chain(set.motion.into_iter().map(|(_, _, f)| f));
1060    for f in frames {
1061        if spare.len() >= KEEP {
1062            return;
1063        }
1064        if let Ok(f) = Arc::try_unwrap(f) {
1065            spare.push(f);
1066        }
1067    }
1068}

1070fn audio_thread(shared: Arc<Shared>, rx: Receiver<Cmd>, backend: Backend) {
1071    use cpal::traits::StreamTrait;
1072    let mut pool = DecoderPool::new(backend);
1073    let mut mixer = Mixer::new();
1074    let ring: Arc<Mutex<VecDeque<f32>>> = Arc::new(Mutex::new(VecDeque::new()));
1075    let dead = Arc::new(AtomicBool::new(false)); // set by the stream's error callback (device gone/changed)
1076    let mut stream: Option<cpal::Stream> = None; // opened on the first Play, re-opened once `dead`
1077    let mut running = false; // device stream started (kept stopped while idle: no wakeups, no audiodg load)
1078    let mut project = lock(&shared.project).clone();
1079    let mut block = vec![0f32; BLOCK * 2];
1080    let mut mixed_until = 0.0;
1081    // true from a Play/Seek reset until the ring has been pre-filled to LEAD_SECS once: suppresses the
1082    // underrun catch-up below so a slow first block (cold device/decoder open) cannot skip mixed_until
1083    // past a fade-in sitting at the new start time — it only ever plays audio mixed from that exact time.
1084    let mut filling = false;
1085    let mut pending: Option<Cmd> = None;
1086    let is_playing = || {
1087        let mut c = lock(&shared.clock);
1088        c.now();
1089        c.playing
1090    };
1091    loop {
1092        if pending.is_none() && !running && !(stream.is_some() && is_playing()) {
1093            let Ok(c) = rx.recv() else { return }; // idle (or no device): block, zero CPU
1094            pending = Some(c);
1095        }
1096        loop {
1097            let c = match pending.take() {
1098                Some(c) => c,
1099                None => match rx.try_recv() {
1100                    Ok(c) => c,
1101                    Err(TryRecvError::Empty) => break,
1102                    Err(TryRecvError::Disconnected) => return,
1103                },
1104            };
1105            match c {
1106                Cmd::SetProject => project = lock(&shared.project).clone(),
1107                Cmd::Play | Cmd::Seek => {
1108                    lock(&ring).clear();
1109                    mixed_until = lock(&shared.clock).now();
1110                    filling = true;
1111                }
1112                Cmd::Pause => lock(&ring).clear(),
1113                Cmd::Canvas => {}
1114                Cmd::Backend(b) => pool.set_backend(b),
1115                Cmd::Gpu(_) | Cmd::Proxies(_) => {} // render-thread only
1116                Cmd::ClearDecoders(ack) => {
1117                    pool.clear();
1118                    let _ = ack.send(());
1119                }
1120                Cmd::RenderOnce(..) | Cmd::LayersOnce(..) => {} // render-thread only
1121                Cmd::Quit => return,
1122            }
1123        }
1124        let playing = is_playing();
1125        if playing && (stream.is_none() || dead.swap(false, Ordering::Relaxed)) {
1126            // lazy/late open: no device at startup, or it went away / the default changed mid-session.
1127            // If it fails we fall back to the idle recv above (retried on the next command).
1128            stream = open_output(ring.clone(), dead.clone());
1129            running = false;
1130        }
1131        let Some(stream) = &stream else { continue };
1132        if playing && !running {
1133            running = stream.play().is_ok();
1134        } else if !playing && running && lock(&ring).is_empty() {
1135            // let the tail play out after the clock auto-stops at the end, then stop the device
1136            let _ = stream.pause();
1137            running = false;
1138        }
1139        let queued = lock(&ring).len() as f64 / (2 * SAMPLE_RATE) as f64;
1140        if playing && queued < LEAD_SECS {
1141            // ponytail: wall clock is the master; the ring is refilled only as the device drains it, so it
1142            // can't grow past LEAD + one block. The ring head plays at timeline time `mixed_until - queued`;
1143            // if that is already >50 ms in the past (decoder respawn, mid-playback underrun) snap the next
1144            // block to land on time instead of carrying the lag forever. Not while still `filling`, though:
1145            // a cold device/decoder open makes this same block "late" on every fresh Play/Seek, and snapping
1146            // then would skip mixed_until straight past a fade-in sitting at the new start time. Resample if
1147            // drift ever matters.
1148            let now = lock(&shared.clock).now();
1149            if !filling && now - (mixed_until - queued) > 0.05 {
1150                mixed_until = now + queued;
1151            }
1152            if guarded(&mut pool, |pool| mixer.mix(&project, mixed_until, pool, &mut block)) {
1153                lock(&ring).extend(block.iter().copied()); // a panicked block would be garbage: underrun instead
1154            }
1155            mixed_until += BLOCK as f64 / SAMPLE_RATE as f64;
1156        } else {
1157            filling = false; // ring reached LEAD_SECS (or we're paused): back to normal underrun detection
1158            match rx.recv_timeout(Duration::from_millis(4)) {
1159                Ok(c) => pending = Some(c),
1160                Err(RecvTimeoutError::Disconnected) => return,
1161                Err(RecvTimeoutError::Timeout) => {}
1162            }
1163        }
1164    }
1165}

1167/// Default output device as 48 kHz stereo f32; None = no audio (playback still runs off the wall clock).
1168fn open_output(ring: Arc<Mutex<VecDeque<f32>>>, dead: Arc<AtomicBool>) -> Option<cpal::Stream> {
1169    use cpal::traits::{DeviceTrait, HostTrait};
1170    let device = cpal::default_host().default_output_device()?;
1171    // ponytail: WASAPI shared mode converts any PCM rate/channel count for output (cpal sets
1172    // AUTOCONVERTPCM), so we ask for exactly what the mixer produces. Add a resampler if a device refuses.
1173    let config = cpal::StreamConfig { channels: 2, sample_rate: SAMPLE_RATE, buffer_size: cpal::BufferSize::Default };
1174    let stream = device
1175        .build_output_stream(
1176            config,
1177            move |out: &mut [f32], _: &cpal::OutputCallbackInfo| {
1178                let mut r = lock(&ring);
1179                for s in out.iter_mut() {
1180                    *s = r.pop_front().unwrap_or(0.0); // zero-fill on underrun
1181                }
1182            },
1183            move |e| {
1184                eprintln!("audio: {e}");
1185                // cpal keeps the stream on the old endpoint after a default-device change and only reports
1186                // it; any non-xrun error means "rebuild the stream" (done lazily by audio_thread).
1187                if e.kind() != cpal::ErrorKind::Xrun {
1188                    dead.store(true, Ordering::Relaxed);
1189                }
1190            },
1191            None,
1192        )
1193        .map_err(|e| eprintln!("audio: no output stream ({e}); playing silently"))
1194        .ok()?;
1195    Some(stream)
1196}
