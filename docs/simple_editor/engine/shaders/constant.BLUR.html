            acc += texture(tex, uv + vec2(cos(ang), sin(ang)) * rad / u_res) * w;
            wsum += w;
        }
        res = acc / wsum;

    float m = u_has_mask > 0.5 ? texture(u_mask, uv).r : 1.0;
